from pathlib import Path

def search_directory(directory: Path, name_parts: list\\[str\\]) -> tuple\\[list, list, bool\\]:

    \"\"\"

    Search only the immediate directory level for files containing all name parts.

    

    Returns:

        tuple of (found_paths, invalid_paths, found_match)

    \"\"\"

    found_paths = \\[\\]

    invalid_paths = \\[\\]

    

    try:

        name_parts_lower = \\[part.lower() for part in name_parts\\]

        

        # Only check files in the immediate directory

        files = \\[f for f in directory.iterdir() if f.is_file()\\]

        

        # Check for matching files

        for file in files:

            if all(part in file.name.lower() for part in name_parts_lower):

                found_paths.append(str(file.absolute()))

                return found_paths, invalid_paths, True  # Found match, skip subdirectories

        

        if not any(d.is_dir() for d in directory.iterdir()):

            invalid_paths.append(str(directory.absolute()))

                

    except PermissionError:

        print(f\"Permission denied: {directory}\")

    except Exception as e:

        print(f\"Error in {directory}: {e}\")

        

    return found_paths, invalid_paths, False



def validate_files(search_terms: str, root_path: str) -> None:

    \"\"\"

    Find all directories that contain matching files at their immediate level,

    excluding subdirectories of directories where matches are found.

    \"\"\"

    name_parts = \\[term.strip() for term in search_terms.split() if term.strip()\\]

    

    root = Path(root_path).expanduser()

    if not root.exists():

        print(f\"Path not found: {root_path}\")

        return



    found_paths = \\[\\]

    invalid_paths = \\[\\]

    skip_dirs = set()  # Directories to skip (subdirectories of matches)

    

    # Get and sort directories by depth to process parents first

    try:

        all_dirs = \\[root\\] + \\[d for d in root.rglob('\\*') if d.is_dir()\\]

        directories = sorted(all_dirs, key=lambda x: len(x.parts))

    except PermissionError:

        print(f\"Permission denied accessing {root_path}\")

        return

    except Exception as e:

        print(f\"Error accessing {root_path}: {e}\")

        return



    # Search each directory at its immediate level

    for directory in directories:

        # Skip if this is a subdirectory of a directory where we found a match

        if any(str(directory).startswith(str(skip)) for skip in skip_dirs):

            continue

            

        found, invalid, found_match = search_directory(directory, name_parts)

        if found_match:

            found_paths.extend(found)

            skip_dirs.add(directory)  # Add this directory to skip list

        invalid_paths.extend(invalid)



    # Sort results

    found_paths.sort()

    invalid_paths.sort()

    

    # Group results by directory

    directories_with_matches = {}

    for path in found_paths:

        dir_path = str(Path(path).parent)

        if dir_path not in directories_with_matches:

            directories_with_matches\\[dir_path\\] = Path(path).name

    

    # Print results

    print(\"\\\
\" + \"=\"\\*80)

    print(f\"VALIDATION RESULTS FOR FILES CONTAINING: '{search_terms}'\")

    print(\"=\"\\*80)

    

    print(f\"\\\
DIRECTORIES WITH MATCHING FILES ({len(directories_with_matches)}):\")

    print(\"-\" \\* 40)

    for directory, filename in sorted(directories_with_matches.items()):

        print(f\"✓ {directory}\")

        print(f\"  Found: {filename}\")

    

    print(f\"\\\
LEAF DIRECTORIES WITHOUT MATCHES ({len(invalid_paths)}):\")

    print(\"-\" \\* 40)

    for path in invalid_paths:

        if not any(str(path).startswith(str(skip)) for skip in skip_dirs):

            print(f\"✗ {path}\")

    

    print(\"\\\
\" + \"=\"\\*80)

    print(\"SUMMARY\")

    print(\"=\"\\*80)

    print(f\"Directories containing matches: {len(directories_with_matches)}\")

    print(f\"Leaf directories without matches: {len(invalid_paths)}\")



def main():

    while True:

        root_path = input(\"Enter the root directory path: \").strip()

        try:

            path = Path(root_path)

            if path.exists() and path.is_dir():

                break

            print(\"Invalid path: Directory does not exist\")

        except Exception as e:

            print(f\"Invalid path: {e}\")

    

    search_terms = input(\"Enter search terms (space-separated): \").strip()

    

    print(\"\\\
Scanning directories... This might take a while for large directory structures.\")

    

    validate_files(search_terms, root_path)



if \\__name_\\_ == \"\\__main_\\_\":
